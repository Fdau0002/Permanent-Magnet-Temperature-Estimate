pmsm_project.features

Paper-inspired predictors and profile-local finite-history exponential moments.

All spans are sample counts at 2 Hz. Neither target temperature nor profile ID enters the 108 predictors. The vectorised sampler deliberately retains its original span-major insertion order. Stage 5 retains that order consistently. Stage 6/Stage 9 explicitly select variable-major canonical order; the final feature_frame supplies it directly. Model schemas are not mixed. See Kirchgässner et al. (2021).

 16def make_paper_inputs(data: pd.DataFrame) -> pd.DataFrame:
 17    """Construct the 12 instantaneous paper-inspired PMSM predictors.
 18
 19    Parameters
 20    ----------
 21    data : pandas.DataFrame
 22        Original PMSM measurements, with ambient, coolant, u_d, u_q,
 23        i_d, i_q and motor_speed in the dataset's original units.
 24
 25    Returns
 26    -------
 27    pandas.DataFrame
 28        Twelve predictors, in the original row order and index; no
 29        target, session identifier, or downstream fitted parameters.
 30
 31    Raises
 32    ------
 33    ValueError
 34        If one of the seven measured input columns is missing.
 35
 36    Notes
 37    -----
 38    Ambient/coolant temperatures use deg C, dq voltages V, dq currents A,
 39    and speed rpm. Derived magnitudes, apparent-power proxy and speed
 40    interactions inherit the corresponding compound units. Sensor
 41    calibration and physically safe operating limits are not inferred.
 42    """
 43    required = ("ambient", "coolant", "u_d", "u_q", "i_d", "i_q", "motor_speed")
 44    missing = sorted(set(required) - set(data.columns))
 45    if missing:
 46        raise ValueError(f"Missing measured inputs: {missing}")
 47
 48    X = pd.DataFrame(index=data.index)
 49    X["theta_a"] = data["ambient"]
 50    X["theta_c"] = data["coolant"]
 51    X["u_d"] = data["u_d"]
 52    X["u_q"] = data["u_q"]
 53    X["i_d"] = data["i_d"]
 54    X["i_q"] = data["i_q"]
 55    X["n_mech"] = data["motor_speed"]
 56
 57    X["u_s"] = np.hypot(X["u_d"], X["u_q"])
 58    X["i_s"] = np.hypot(X["i_d"], X["i_q"])
 59    X["S_el"] = 1.5 * X["u_s"] * X["i_s"]
 60    omega = (2 * np.pi / 60) * X["n_mech"]
 61    X["i_s_omega"] = X["i_s"] * omega
 62    X["S_el_omega"] = X["S_el"] * omega
 63
 64    expected = (
 65        "theta_a",
 66        "theta_c",
 67        "u_d",
 68        "u_q",
 69        "i_d",
 70        "i_q",
 71        "n_mech",
 72        "u_s",
 73        "i_s",
 74        "S_el",
 75        "i_s_omega",
 76        "S_el_omega",
 77    )
 78    assert tuple(X.columns) == expected
 79    assert "pm" not in X.columns and "profile_id" not in X.columns
 80    return X

 83def make_temporal_features(
 84    X: pd.DataFrame, profile_ids: pd.Series, spans=(10, 50, 200, 1000)
 85) -> pd.DataFrame:
 86    """Add causal EWMA and EWMS features, restarting at each profile.
 87
 88    Parameters
 89    ----------
 90    X : pandas.DataFrame
 91        Numeric predictors in chronological (original dataset row) order.
 92    profile_ids : pandas.Series
 93        Recording-session IDs aligned with X.index. Each session must
 94        appear in a single chronologically ordered block.
 95    spans : sequence of positive integers
 96        EWMA smoothing spans in sample counts (not seconds). These are
 97        illustrative until cross-validated in the modelling stage.
 98
 99    Returns
100    -------
101    pandas.DataFrame
102        Original columns followed by EWMA/EWMS pairs for each input
103        and span; 12 + 2*12*4 = 108 columns for the default values.
104
105    Raises
106    ------
107    TypeError
108        If the predictors/IDs are not a DataFrame and aligned Series.
109    ValueError
110        If input is empty, missing/nonfinite, misaligned, nonchronological,
111        interleaved across sessions, or spans are not distinct positive integers.
112
113    Notes
114    -----
115    adjust=True implements a finite-history, normalized EW mean.
116    std(bias=True) produces its population weighted standard deviation.
117    These are feature transformations, not train-fitted scalers.
118    """
119    if not isinstance(X, pd.DataFrame) or not isinstance(profile_ids, pd.Series):
120        raise TypeError("X must be a DataFrame and profile_ids must be a Series")
121    if not X.index.equals(profile_ids.index):
122        raise ValueError("X and profile_ids must have exactly aligned indices")
123    if not X.index.is_unique or X.empty or profile_ids.isna().any():
124        raise ValueError(
125            "Require unique row index, nonempty data, and valid profile IDs"
126        )
127    if not X.index.is_monotonic_increasing:
128        raise ValueError("Preserve increasing original source-row order")
129    if int(profile_ids.ne(profile_ids.shift()).sum()) != profile_ids.nunique():
130        raise ValueError("Each profile must occupy one contiguous block")
131    spans = tuple(spans)
132    if (
133        not spans
134        or len(set(spans)) != len(spans)
135        or any(not isinstance(s, (int, np.integer)) or s < 1 for s in spans)
136    ):
137        raise ValueError("Spans must be distinct positive integers")
138    if X.isna().any().any() or not np.isfinite(X.to_numpy()).all():
139        raise ValueError("Inputs must have finite nonmissing values")
140
141    # These features are calculated within a profile, in existing row order.
142    # Note that the user must preserve chronological order before calling.
143    engineered = {}
144    for column in X.columns:
145        grouped = X[column].groupby(profile_ids, sort=False)
146        for s in spans:
147            ewma_name = f"{column}_ewma_s{s}"
148            ewms_name = f"{column}_ewms_s{s}"
149            engineered[ewma_name] = grouped.transform(
150                lambda v: v.ewm(span=s, adjust=True).mean()
151            )
152            engineered[ewms_name] = grouped.transform(
153                lambda v: v.ewm(span=s, adjust=True).std(bias=True)
154            )
155
156    engineered_frame = pd.DataFrame(engineered, index=X.index)
157    return pd.concat([X, engineered_frame], axis=1)

160def vectorised_temporal_sample(
161    X_original: pd.DataFrame, positions: np.ndarray, spans: tuple | list
162) -> pd.DataFrame:
163    """Sample profile-local exponential moments after complete-history filtering.
164
165    Parameters
166    ----------
167    X_original : pandas.DataFrame
168        Twelve instantaneous predictors for one recording in original row order.
169        Mixed physical units are inherited from ``make_paper_inputs``.
170    positions : array-like of int
171        Zero-based within-profile positions, selected after filtering.
172    spans : sequence of int
173        Four smoothing spans in samples at 2 Hz; adjust=True and bias=True.
174
175    Returns
176    -------
177    pandas.DataFrame
178        Selected rows, indexed by original source row. Columns are base inputs
179        followed by span-major EWMA/EWMS pairs. Use ``stage6_feature_names``
180        to select the canonical variable-major model order.
181
182    Raises
183    ------
184    IndexError
185        If a position is outside the profile.
186    ValueError
187        If pandas cannot interpret a smoothing span.
188
189    Notes
190    -----
191    No fitted normalisation is performed. The caller must supply one complete
192    chronological profile; passing disjoint or shuffled histories changes the
193    meaning of the features. The original sampler is retained verbatim."""
194    sampled_columns = {
195        col: X_original[col].iloc[positions].to_numpy() for col in X_original.columns
196    }
197    for s in spans:
198        ewm = X_original.ewm(span=int(s), adjust=True)
199        historical_mean = ewm.mean().iloc[positions]
200        historical_std = ewm.std(bias=True).iloc[positions]
201        for col in X_original.columns:
202            sampled_columns[f"{col}_ewma_s{s}"] = historical_mean[col].to_numpy()
203            sampled_columns[f"{col}_ewms_s{s}"] = historical_std[col].to_numpy()
204    return pd.DataFrame(sampled_columns, index=X_original.index[positions])

207def stage6_feature_names(columns: list | tuple, spans: list | tuple) -> list[str]:
208    """Construct the development canonical model feature order.
209
210    Parameters
211    ----------
212    columns : sequence of str
213        Ordered instantaneous input names (normally twelve).
214    spans : sequence of int
215        Ordered spans in samples; normally four.
216
217    Returns
218    -------
219    list of str
220        Raw names then EWMA/EWMS pairs per variable and span, normally 108.
221
222    Notes
223    -----
224    Column ordering is part of a fitted model's interface. This function does
225    not sort spans or infer units. It deliberately preserves the caller's order."""
226    return list(columns) + [
227        name
228        for col in columns
229        for span in spans
230        for name in (f"{col}_ewma_s{span}", f"{col}_ewms_s{span}")
231    ]

234def feature_names(spans: list | tuple) -> list[str]:
235    """Return the final estimator's canonical 108-column interface.
236
237    Parameters
238    ----------
239    spans : sequence of int
240        Four frozen smoothing spans in samples.
241
242    Returns
243    -------
244    list of str
245        Twelve base names plus paired temporal moments in model order.
246
247    Notes
248    -----
249    Feature moments retain the units of each base predictor. Names must match
250    both persisted model metadata and inference matrices exactly."""
251    return BASE_COLS + [
252        key
253        for col in BASE_COLS
254        for s in spans
255        for key in (f"{col}_ewma_s{s}", f"{col}_ewms_s{s}")
256    ]

259def feature_frame(profile: pd.DataFrame, spans: list | tuple) -> pd.DataFrame:
260    """Compute the original final-notebook feature matrix for one session.
261
262    Parameters
263    ----------
264    profile : pandas.DataFrame
265        Exactly one recording, in increasing original source-row order.
266        Seven sensor columns plus ``profile_id`` are required.
267    spans : sequence of int
268        Four frozen exponential smoothing spans (samples).
269
270    Returns
271    -------
272    pandas.DataFrame
273        Finite float-valued matrix with 108 columns in canonical order.
274
275    Raises
276    ------
277    AssertionError
278        If profiles mix, ordering is invalid, shape differs or values are nonfinite.
279    KeyError
280        If required measured columns are absent.
281
282    Notes
283    -----
284    Every EWMA uses adjust=True and every EWMS uses bias=True. The initial
285    mean is the first observation and the initial population standard deviation
286    is zero. Filtering uses current and previous inputs, never future values.
287    Assertions retain the supplied notebook's valid-input behaviour. Do not run
288    Python with -O because experimental assertions are part of its audit."""
289    assert (
290        profile.profile_id.nunique() == 1
291    ), "Never mix recording sessions before temporal feature construction."
292    assert (
293        profile.index.is_monotonic_increasing
294    ), "Original chronological ordering is mandatory."
295    base = make_paper_inputs(profile)
296    out = {}
297    for col in BASE_COLS:
298        for s in spans:
299            ew = base[col].ewm(span=int(s), adjust=True)
300            out[f"{col}_ewma_s{s}"] = ew.mean().to_numpy()
301            out[f"{col}_ewms_s{s}"] = ew.std(bias=True).to_numpy()
302    extra = pd.DataFrame(out, index=base.index)
303    result = pd.concat([base, extra], axis=1)
304    assert list(result.columns) == feature_names(spans)
305    assert result.shape == (len(profile), 108) and np.isfinite(result.to_numpy()).all()
306    return result
