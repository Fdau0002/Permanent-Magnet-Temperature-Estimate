pmsm_project.study

Recovered development objectives with explicit development-only state.

The numerical statements come from the supplied historical reconstruction; provenance/objective_extraction.json records the source cells. This module does not read a saved trial file or a historical optimum.

 27class DevelopmentStudy:
 28    """Bind unchanged numerical objectives to a verified development partition.
 29
 30    Parameters
 31    ----------
 32    development : pandas.DataFrame
 33        Chronological raw measurements; target ``pm`` in deg C.
 34    profile_cv : pandas.DataFrame
 35        One row per profile with ``profile_id`` and integer ``cv_fold`` (1--3).
 36    strict : bool, default True
 37        Enforce original 55 profiles and 1,055,198 rows. False is exclusively
 38        for labelled synthetic numerical tests, never production selection.
 39
 40    Raises
 41    ------
 42    ValueError
 43        On overlapping held-out IDs, missing folds or incompatible partitions.
 44
 45    Examples
 46    --------
 47    >>> # study = DevelopmentStudy(development, profile_cv)
 48    >>> # mean_mse, fold_mse = study.evaluate_ols_cv([10, 50, 200, 1000])
 49    """

 51    def __init__(self, development, profile_cv, strict=True):
 52        ids = set(map(int, development.profile_id))
 53        if ids & set(LOCKED_TEST_IDS):
 54            raise ValueError(
 55                "Held-out profiles are forbidden in development objectives"
 56            )
 57        if (
 58            profile_cv.profile_id.duplicated().any()
 59            or ids != set(profile_cv.profile_id)
 60            or set(profile_cv.cv_fold) != {1, 2, 3}
 61        ):
 62            raise ValueError(
 63                "Require exactly one of three CV folds for every development profile"
 64            )
 65        if strict and (
 66            sorted(ids) != LOCKED_DEVELOPMENT_IDS or len(development) != 1055198
 67        ):
 68            raise ValueError(
 69                "Production optimisation requires all original development profiles/rows"
 70            )
 71        self.development = development
 72        self.profile_cv = profile_cv.copy()
 73        self.fold_lookup = (
 74            profile_cv.set_index("profile_id")["cv_fold"].astype(int).to_dict()
 75        )
 76        self.samples_per_profile = 80
 77        self.train_samples = TRAIN_SAMPLES.copy()
 78        self.validation_samples = VALIDATION_SAMPLES

 80    def build_candidate_features(self, spans):
 81        """Construct features for candidate EWMA/EWMS spans.
 82
 83        Full original 2-Hz histories are processed before
 84        selecting 80 observations per development profile.
 85
 86        Parameters
 87        ----------
 88        spans : sequence[int]
 89            Four ordered causal exponential spans in samples (2 Hz source).
 90
 91        Returns
 92        -------
 93        tuple[numpy.ndarray, numpy.ndarray, numpy.ndarray]
 94            Sampled predictors (mixed physical units), target (deg C), profile IDs.
 95
 96        Raises
 97        ------
 98        ValueError
 99            If candidate parameters, profile lengths or estimator settings are invalid.
100        AssertionError
101            If finite-value, expected-size or profile-isolation invariants fail.
102
103        Notes
104        -----
105        Source arithmetic is retained. Features use complete chronological profiles
106        before sampling; all scalers are fitted within the fitting fold. Fold MSE
107        is in original temperature units after inverse transformation. QR solves
108        can be rank deficient; fitted rank is recorded instead of forming inverses.
109        """
110        feature_arrays = []
111        target_arrays = []
112        profile_arrays = []
113        for pid, profile in self.development.groupby("profile_id", sort=False):
114            X_original = make_paper_inputs(profile)
115            X_temporal = make_temporal_features(
116                X_original, profile["profile_id"], spans
117            )
118            n = min(self.samples_per_profile, len(profile))
119            positions = np.linspace(0, len(profile) - 1, n, dtype=int)
120            feature_arrays.append(X_temporal.iloc[positions].to_numpy(dtype=np.float64))
121            target_arrays.append(profile["pm"].iloc[positions].to_numpy())
122            profile_arrays.append(np.full(n, pid))
123        X = np.vstack(feature_arrays)
124        y = np.concatenate(target_arrays)
125        profile_ids = np.concatenate(profile_arrays)
126        assert X.shape[1] == 108
127        assert len(X) == 4400
128        assert np.isfinite(X).all()
129        return (X, y, profile_ids)

131    def evaluate_ols_cv(self, spans):
132        """Evaluate a four-span configuration with the same
133        fixed, grouped, stratified 3-fold CV.
134
135        Return average CV MSE and individual fold scores.
136
137        Parameters
138        ----------
139        spans : sequence[int]
140            Four ordered causal exponential spans in samples (2 Hz source).
141
142        Returns
143        -------
144        tuple[float, list[float]]
145            Arithmetic mean and three fold MSE scores (deg C squared).
146
147        Raises
148        ------
149        ValueError
150            If candidate parameters, profile lengths or estimator settings are invalid.
151        AssertionError
152            If finite-value, expected-size or profile-isolation invariants fail.
153
154        Notes
155        -----
156        Source arithmetic is retained. Features use complete chronological profiles
157        before sampling; all scalers are fitted within the fitting fold. Fold MSE
158        is in original temperature units after inverse transformation. QR solves
159        can be rank deficient; fitted rank is recorded instead of forming inverses.
160        """
161        X, y, profile_ids = self.build_candidate_features(spans)
162        fold_ids = pd.Series(profile_ids).map(self.fold_lookup).to_numpy(dtype=int)
163        fold_mse = []
164        for fold in [1, 2, 3]:
165            train_mask = fold_ids != fold
166            val_mask = fold_ids == fold
167            assert train_mask.sum() > 0
168            assert val_mask.sum() > 0
169            assert set(profile_ids[train_mask]).isdisjoint(set(profile_ids[val_mask]))
170            pipeline = Pipeline(
171                [("scaler", StandardScaler()), ("model", LinearRegression())]
172            )
173            pipeline.fit(X[train_mask], y[train_mask])
174            predictions = pipeline.predict(X[val_mask])
175            mse = mean_squared_error(y[val_mask], predictions)
176            fold_mse.append(float(mse))
177        return (float(np.mean(fold_mse)), fold_mse)

179    def evaluate_model_cv(self, model_name, parameters):
180        """Evaluate model hyperparameters on the fixed
181        stratified grouped 3-fold cross-validation split.
182
183        Parameters
184        ----------
185        model_name : str
186            SVR, k-NN, Random Forest, Extra Trees or MLP.
187        parameters : sequence
188            First span, three gaps, then model parameters in search-space order.
189
190        Returns
191        -------
192        dict
193            Spans, hyperparameters, fold MSE, mean MSE and convergence count.
194
195        Raises
196        ------
197        ValueError
198            If candidate parameters, profile lengths or estimator settings are invalid.
199        AssertionError
200            If finite-value, expected-size or profile-isolation invariants fail.
201
202        Notes
203        -----
204        Source arithmetic is retained. Features use complete chronological profiles
205        before sampling; all scalers are fitted within the fitting fold. Fold MSE
206        is in original temperature units after inverse transformation. QR solves
207        can be rank deficient; fitted rank is recorded instead of forming inverses.
208        """
209        spans = decode_spans(parameters[:4])
210        hp = {
211            dimension.name: value
212            for dimension, value in zip(MODEL_SPACES[model_name], parameters[4:])
213        }
214        X, y, profile_ids = self.build_candidate_features(spans)
215        fold_lookup = self.profile_cv.set_index("profile_id")["cv_fold"]
216        fold_ids = pd.Series(profile_ids).map(self.fold_lookup).to_numpy()
217        assert not pd.isna(fold_ids).any()
218        fold_scores = []
219        convergence_warnings = 0
220        for fold in [1, 2, 3]:
221            training_mask = fold_ids != fold
222            validation_mask = fold_ids == fold
223            assert set(profile_ids[training_mask]).isdisjoint(
224                set(profile_ids[validation_mask])
225            )
226            estimator = make_regressor(model_name, hp)
227            feature_pipeline = Pipeline(
228                [("scaler", StandardScaler()), ("model", estimator)]
229            )
230            pipeline = TransformedTargetRegressor(
231                regressor=feature_pipeline, transformer=StandardScaler()
232            )
233            with warnings.catch_warnings(record=True) as caught:
234                warnings.simplefilter("always", ConvergenceWarning)
235                pipeline.fit(X[training_mask], y[training_mask])
236            convergence_warnings += sum(
237                (issubclass(w.category, ConvergenceWarning) for w in caught)
238            )
239            predictions = pipeline.predict(X[validation_mask])
240            if not np.isfinite(predictions).all():
241                raise ValueError(f"{model_name} produced invalid predictions")
242            mse = mean_squared_error(y[validation_mask], predictions)
243            fold_scores.append(float(mse))
244        return {
245            "spans": spans,
246            "hyperparameters": hp,
247            "fold_scores": fold_scores,
248            "mean_cv_mse": float(np.mean(fold_scores)),
249            "convergence_warnings": convergence_warnings,
250        }

252    def evaluate_full_ols_cv(self, spans):
253        """Evaluate an OLS configuration on all development rows.
254
255        Uses profile-wise QR compression and training-fold-only
256        standardisation. No test profiles are accessed.
257
258        Parameters
259        ----------
260        spans : sequence[int]
261            Four ordered causal exponential spans in samples (2 Hz source).
262
263        Returns
264        -------
265        dict
266            Spans, fold table with numerical rank/MSE, and measured timing (seconds).
267
268        Raises
269        ------
270        ValueError
271            If candidate parameters, profile lengths or estimator settings are invalid.
272        AssertionError
273            If finite-value, expected-size or profile-isolation invariants fail.
274
275        Notes
276        -----
277        Source arithmetic is retained. Features use complete chronological profiles
278        before sampling; all scalers are fitted within the fitting fold. Fold MSE
279        is in original temperature units after inverse transformation. QR solves
280        can be rank deficient; fitted rank is recorded instead of forming inverses.
281        """
282        start_total = time.perf_counter()
283        profile_stats = {}
284        for count, (pid, profile) in enumerate(
285            self.development.groupby("profile_id", sort=False), start=1
286        ):
287            pid = int(pid)
288            profile_stats[pid] = compress_profile_ols(profile, spans)
289            if count % 10 == 0 or count == 55:
290                print(f"Processed {count}/55 profiles", flush=True)
291        feature_time = time.perf_counter() - start_total
292        start_cv = time.perf_counter()
293        fold_results = []
294        for fold in [1, 2, 3]:
295            training_ids = [
296                pid for pid in profile_stats if self.fold_lookup[pid] != fold
297            ]
298            validation_ids = [
299                pid for pid in profile_stats if self.fold_lookup[pid] == fold
300            ]
301            assert set(training_ids).isdisjoint(set(validation_ids))
302            n_train = sum((profile_stats[pid]["n"] for pid in training_ids))
303            mu = (
304                sum(
305                    (
306                        profile_stats[pid]["n"] * profile_stats[pid]["mean"]
307                        for pid in training_ids
308                    )
309                )
310                / n_train
311            )
312            variance = (
313                sum(
314                    (
315                        profile_stats[pid]["n"]
316                        * (
317                            profile_stats[pid]["var"]
318                            + (profile_stats[pid]["mean"] - mu) ** 2
319                        )
320                        for pid in training_ids
321                    )
322                )
323                / n_train
324            )
325            scale = np.sqrt(np.maximum(variance, 0))
326            scale[scale == 0] = 1.0
327            R_train = np.vstack([profile_stats[pid]["R"] for pid in training_ids])
328            R_scaled = R_train.copy()
329            R_scaled[:, 1:-1] = (R_train[:, 1:-1] - R_train[:, [0]] * mu) / scale
330            coefficients, _, rank, singular_values = lstsq(
331                R_scaled[:, :-1], R_scaled[:, -1], lapack_driver="gelsd"
332            )
333            beta = coefficients[1:] / scale
334            intercept = coefficients[0] - np.dot(mu, beta)
335            residual_vector = np.concatenate([[intercept], beta, [-1.0]])
336            n_validation = 0
337            validation_sse = 0.0
338            for pid in validation_ids:
339                R = profile_stats[pid]["R"]
340                residuals_compressed = R @ residual_vector
341                validation_sse += float(
342                    np.dot(residuals_compressed, residuals_compressed)
343                )
344                n_validation += profile_stats[pid]["n"]
345            mse = validation_sse / n_validation
346            assert np.isfinite(mse)
347            assert mse >= 0
348            fold_results.append(
349                {
350                    "Fold": fold,
351                    "Training_Rows": n_train,
352                    "Validation_Rows": n_validation,
353                    "MSE": mse,
354                    "Numerical_Rank": rank,
355                }
356            )
357        cv_time = time.perf_counter() - start_cv
358        results = pd.DataFrame(fold_results)
359        mean_cv_mse = results["MSE"].mean()
360        return {
361            "spans": list(map(int, spans)),
362            "fold_results": results,
363            "mean_cv_mse": float(mean_cv_mse),
364            "feature_time_s": feature_time,
365            "cv_time_s": cv_time,
366            "total_time_s": time.perf_counter() - start_total,
367        }

369    def evaluate_larger_model_cv(self, model_name, parameters):
370        """Evaluate one Bayesian candidate using:
371        - Full original 2-Hz temporal histories
372        - Nested training samples
373        - Fixed 640-sample/profile validation
374        - Three profile-grouped CV folds
375
376        Parameters
377        ----------
378        model_name : str
379            SVR, k-NN, Random Forest, Extra Trees or MLP.
380        parameters : sequence
381            First span, three gaps, then model parameters in search-space order.
382
383        Returns
384        -------
385        dict
386            Selected candidate, three MSE values, timing and convergence count.
387
388        Raises
389        ------
390        ValueError
391            If candidate parameters, profile lengths or estimator settings are invalid.
392        AssertionError
393            If finite-value, expected-size or profile-isolation invariants fail.
394
395        Notes
396        -----
397        Source arithmetic is retained. Features use complete chronological profiles
398        before sampling; all scalers are fitted within the fitting fold. Fold MSE
399        is in original temperature units after inverse transformation. QR solves
400        can be rank deficient; fitted rank is recorded instead of forming inverses.
401        """
402        start_total = time.perf_counter()
403        n_train = self.train_samples[model_name]
404        n_val = self.validation_samples
405        n_select = max(n_train, n_val)
406        spans = decode_spans(parameters[:4])
407        hp = {
408            dimension.name: value
409            for dimension, value in zip(MODEL_SPACES[model_name], parameters[4:])
410        }
411        feature_arrays = []
412        target_arrays = []
413        profile_arrays = []
414        rank_arrays = []
415        for pid, profile in self.development.groupby("profile_id", sort=False):
416            order = nested_sample_order(len(profile), n_select)
417            positions = np.sort(np.asarray(order, dtype=int))
418            rank_lookup = {int(position): rank for rank, position in enumerate(order)}
419            ranks = np.asarray([rank_lookup[int(position)] for position in positions])
420            X_original = make_paper_inputs(profile)
421            X_full = make_temporal_features(X_original, profile["profile_id"], spans)
422            X_selected = X_full.iloc[positions].to_numpy(dtype=np.float64)
423            y_selected = profile["pm"].iloc[positions].to_numpy(dtype=np.float64)
424            feature_arrays.append(X_selected)
425            target_arrays.append(y_selected)
426            profile_arrays.append(np.full(len(positions), int(pid)))
427            rank_arrays.append(ranks)
428        X = np.vstack(feature_arrays)
429        y = np.concatenate(target_arrays)
430        profile_ids = np.concatenate(profile_arrays)
431        sample_ranks = np.concatenate(rank_arrays)
432        feature_time = time.perf_counter() - start_total
433        assert X.shape[1] == 108
434        assert np.isfinite(X).all()
435        assert np.isfinite(y).all()
436        fold_ids = pd.Series(profile_ids).map(self.fold_lookup).to_numpy(dtype=int)
437        fold_scores = []
438        convergence_warnings = 0
439        training_time = 0.0
440        inference_time = 0.0
441        for fold in [1, 2, 3]:
442            train_mask = (fold_ids != fold) & (sample_ranks < n_train)
443            val_mask = (fold_ids == fold) & (sample_ranks < n_val)
444            assert set(profile_ids[train_mask]).isdisjoint(set(profile_ids[val_mask]))
445            estimator = make_regressor(model_name, hp)
446            if model_name == "MLP":
447                estimator.max_iter = 400
448                estimator.early_stopping = True
449                estimator.validation_fraction = 0.1
450                estimator.n_iter_no_change = 15
451            pipeline = TransformedTargetRegressor(
452                regressor=Pipeline(
453                    [("scaler", StandardScaler()), ("model", estimator)]
454                ),
455                transformer=StandardScaler(),
456            )
457            start_fit = time.perf_counter()
458            with warnings.catch_warnings(record=True) as caught:
459                warnings.simplefilter("always", ConvergenceWarning)
460                pipeline.fit(X[train_mask], y[train_mask])
461            training_time += time.perf_counter() - start_fit
462            convergence_warnings += sum(
463                (issubclass(w.category, ConvergenceWarning) for w in caught)
464            )
465            start_predict = time.perf_counter()
466            predictions = pipeline.predict(X[val_mask])
467            inference_time += time.perf_counter() - start_predict
468            assert np.isfinite(predictions).all()
469            fold_scores.append(float(mean_squared_error(y[val_mask], predictions)))
470        mean_mse = float(np.mean(fold_scores))
471        return {
472            "spans": spans,
473            "hyperparameters": hp,
474            "fold_scores": fold_scores,
475            "mean_cv_mse": mean_mse,
476            "feature_time_s": feature_time,
477            "training_time_s": training_time,
478            "inference_time_s": inference_time,
479            "total_time_s": time.perf_counter() - start_total,
480            "convergence_warnings": convergence_warnings,
481        }

253    def evaluate_full_ols_cv(self, spans):
254        """Evaluate an OLS configuration on all development rows.
255
256        Uses profile-wise QR compression and training-fold-only
257        standardisation. No test profiles are accessed.
258
259        Parameters
260        ----------
261        spans : sequence[int]
262            Four ordered causal exponential spans in samples (2 Hz source).
263
264        Returns
265        -------
266        dict
267            Spans, fold table with numerical rank/MSE, and measured timing (seconds).
268
269        Raises
270        ------
271        ValueError
272            If candidate parameters, profile lengths or estimator settings are invalid.
273        AssertionError
274            If finite-value, expected-size or profile-isolation invariants fail.
275
276        Notes
277        -----
278        Source arithmetic is retained. Features use complete chronological profiles
279        before sampling; all scalers are fitted within the fitting fold. Fold MSE
280        is in original temperature units after inverse transformation. QR solves
281        can be rank deficient; fitted rank is recorded instead of forming inverses.
282        """
283        start_total = time.perf_counter()
284        profile_stats = {}
285        for count, (pid, profile) in enumerate(
286            self.development.groupby("profile_id", sort=False), start=1
287        ):
288            pid = int(pid)
289            profile_stats[pid] = compress_profile_ols(profile, spans)
290            if count % 10 == 0 or count == 55:
291                print(f"Processed {count}/55 profiles", flush=True)
292        feature_time = time.perf_counter() - start_total
293        start_cv = time.perf_counter()
294        fold_results = []
295        for fold in [1, 2, 3]:
296            training_ids = [
297                pid for pid in profile_stats if self.fold_lookup[pid] != fold
298            ]
299            validation_ids = [
300                pid for pid in profile_stats if self.fold_lookup[pid] == fold
301            ]
302            assert set(training_ids).isdisjoint(set(validation_ids))
303            n_train = sum((profile_stats[pid]["n"] for pid in training_ids))
304            mu = (
305                sum(
306                    (
307                        profile_stats[pid]["n"] * profile_stats[pid]["mean"]
308                        for pid in training_ids
309                    )
310                )
311                / n_train
312            )
313            variance = (
314                sum(
315                    (
316                        profile_stats[pid]["n"]
317                        * (
318                            profile_stats[pid]["var"]
319                            + (profile_stats[pid]["mean"] - mu) ** 2
320                        )
321                        for pid in training_ids
322                    )
323                )
324                / n_train
325            )
326            scale = np.sqrt(np.maximum(variance, 0))
327            scale[scale == 0] = 1.0
328            R_train = np.vstack([profile_stats[pid]["R"] for pid in training_ids])
329            R_scaled = R_train.copy()
330            R_scaled[:, 1:-1] = (R_train[:, 1:-1] - R_train[:, [0]] * mu) / scale
331            coefficients, _, rank, singular_values = lstsq(
332                R_scaled[:, :-1], R_scaled[:, -1], lapack_driver="gelsd"
333            )
334            beta = coefficients[1:] / scale
335            intercept = coefficients[0] - np.dot(mu, beta)
336            residual_vector = np.concatenate([[intercept], beta, [-1.0]])
337            n_validation = 0
338            validation_sse = 0.0
339            for pid in validation_ids:
340                R = profile_stats[pid]["R"]
341                residuals_compressed = R @ residual_vector
342                validation_sse += float(
343                    np.dot(residuals_compressed, residuals_compressed)
344                )
345                n_validation += profile_stats[pid]["n"]
346            mse = validation_sse / n_validation
347            assert np.isfinite(mse)
348            assert mse >= 0
349            fold_results.append(
350                {
351                    "Fold": fold,
352                    "Training_Rows": n_train,
353                    "Validation_Rows": n_validation,
354                    "MSE": mse,
355                    "Numerical_Rank": rank,
356                }
357            )
358        cv_time = time.perf_counter() - start_cv
359        results = pd.DataFrame(fold_results)
360        mean_cv_mse = results["MSE"].mean()
361        return {
362            "spans": list(map(int, spans)),
363            "fold_results": results,
364            "mean_cv_mse": float(mean_cv_mse),
365            "feature_time_s": feature_time,
366            "cv_time_s": cv_time,
367            "total_time_s": time.perf_counter() - start_total,
368        }

327            scale[scale == 0] = 1.0
328            R_train = np.vstack([profile_stats[pid]["R"] for pid in training_ids])
329            R_scaled = R_train.copy()
330            R_scaled[:, 1:-1] = (R_train[:, 1:-1] - R_train[:, [0]] * mu) / scale
331            coefficients, _, rank, singular_values = lstsq(
332                R_scaled[:, :-1], R_scaled[:, -1], lapack_driver="gelsd"
333            )
334            beta = coefficients[1:] / scale
335            intercept = coefficients[0] - np.dot(mu, beta)
336            residual_vector = np.concatenate([[intercept], beta, [-1.0]])
337            n_validation = 0
338            validation_sse = 0.0
339            for pid in validation_ids:
340                R = profile_stats[pid]["R"]
341                residuals_compressed = R @ residual_vector
342                validation_sse += float(
343                    np.dot(residuals_compressed, residuals_compressed)
344                )
345                n_validation += profile_stats[pid]["n"]
346            mse = validation_sse / n_validation
347            assert np.isfinite(mse)
348            assert mse >= 0
349            fold_results.append(
350                {
351                    "Fold": fold,
352                    "Training_Rows": n_train,
353                    "Validation_Rows": n_validation,
354                    "MSE": mse,
355                    "Numerical_Rank": rank,
356                }
357            )
358        cv_time = time.perf_counter() - start_cv
359        results = pd.DataFrame(fold_results)
360        mean_cv_mse = results["MSE"].mean()
361        return {
362            "spans": list(map(int, spans)),
363            "fold_results": results,
364            "mean_cv_mse": float(mean_cv_mse),
365            "feature_time_s": feature_time,
366            "cv_time_s": cv_time,
367            "total_time_s": time.perf_counter() - start_total,
368        }
